Assignment 4: Dijkstra's Shortest Path (Adjacency List)
Aim: Represent landmarks as vertices and distances as weighted edges. Find the shortest distance from one source landmark to every other landmark.
C++17 Code
#include <iostream> #include <vector> #include <queue> #include <climits> using namespace std;

int main() {
int n = 5;
vector<pair<int, int>> graph[5];

// Add roads: landmark, landmark, distance graph[0].push_back({1, 4});
graph[0].push_back({2, 1});
graph[2].push_back({1, 2});
graph[1].push_back({3, 1});
graph[2].push_back({3, 5});
graph[3].push_back({4, 3});

int source = 0;
vector<int> dist(n, INT_MAX); priority_queue<pair<int, int>,
vector<pair<int, int>>, greater<pair<int, int>>> pq;

dist[source] = 0; pq.push({0, source});

while (!pq.empty()) {
int d = pq.top().first; int u = pq.top().second; pq.pop();
if (d != dist[u]) continue; for (auto edge : graph[u]) {
int v = edge.first;
int w = edge.second;

if (dist[u] + w < dist[v]) { dist[v] = dist[u] + w; pq.push({dist[v], v});
}
}
}

cout << "Shortest distances from landmark " << source << ":\n"; for (int i = 0; i < n; i++) {
cout << source << " to " << i << " = ";
if (dist[i] == INT_MAX) cout << "Not reachable"; else cout << dist[i];
cout << endl;
}
return 0;
}
Sample Output
Shortest distances from landmark 0:\n0 to 0 = 0\n0 to 1 = 3\n0 to 2 = 1\n0 to 3 = 4\n0 to 4 = 7
Time / Space Complexity: O((V + E) log V) using an adjacency list and priority queue.

Assignment 5: Hash Table: Chaining and Linear Probing
Aim: Insert and search keys using two collision-handling methods: separate chaining and linear probing.
C++17 Code
#include <iostream> #include <vector> using namespace std;

const int SIZE = 10;

// Separate chaining vector<int> table1[SIZE];

void insertChain(int key) { int index = key % SIZE;
table1[index].push_back(key);
}

bool searchChain(int key) { int index = key % SIZE;
for (int x : table1[index])
if (x == key) return true; return false;
}

// Linear probing int table2[SIZE];

void insertProbe(int key) { int index = key % SIZE; int start = index;

while (table2[index] != -1) { index = (index + 1) % SIZE; if (index == start) {
cout << "Hash table is full\n"; return;
}
}
table2[index] = key;
}

bool searchProbe(int key) { int index = key % SIZE; int start = index;

while (table2[index] != -1) {
if (table2[index] == key) return true; index = (index + 1) % SIZE;
if (index == start) break;
}
return false;
}

int main() {
for (int i = 0; i < SIZE; i++) table2[i] = -1; int keys[] = {12, 22, 32, 15};
for (int key : keys) { insertChain(key); insertProbe(key);
}

cout << "Chaining: search 22 = "
<< (searchChain(22) ? "Found" : "Not found") << endl;

cout << "Linear probing: search 22 = "
<< (searchProbe(22) ? "Found" : "Not found") << endl;

cout << "\nChaining table:\n"; for (int i = 0; i < SIZE; i++) {
cout << i << ": ";
for (int x : table1[i]) cout << x << " "; cout << endl;
}

cout << "\nLinear probing table:\n"; for (int i = 0; i < SIZE; i++)
cout << i << ": " << table2[i] << endl;

return 0;
}

Sample Output

Time / Space Complexity: Average insert/search: O(1). Worst case: O(n).

Assignment 6: In-order Threaded Binary Search Tree
Aim: Create an in-order threaded binary search tree and display its in-order and pre-order traversals.
C++17 Code
#include <iostream> using namespace std;

struct Node {
int data;
Node *left, *right; bool lthread, rthread;
};

Node* newNode(int value) { Node* p = new Node;
p->data = value;
p->left = p->right = nullptr;
p->lthread = p->rthread = true; return p;
}

Node* insert(Node* root, int value) {
if (root == nullptr) return newNode(value);

Node* cur = root; while (true) {
if (value < cur->data) {
if (!cur->lthread) cur = cur->left; else {
Node* p = newNode(value); p->left = cur->left;
p->right = cur; cur->left = p;
cur->lthread = false; break;
}
} else if (value > cur->data) {
if (!cur->rthread) cur = cur->right; else {
Node* p = newNode(value); p->right = cur->right;
p->left = cur; cur->right = p;
cur->rthread = false; break;
}
} else break;
}
return root;
}

Node* leftMost(Node* p) {
if (p == nullptr) return nullptr; while (!p->lthread) p = p->left; return p;
}

void inorder(Node* root) { Node* cur = leftMost(root); while (cur != nullptr) {
cout << cur->data << " ";
if (cur->rthread) cur = cur->right; else cur = leftMost(cur->right);
}
}

void preorder(Node* root) { Node* cur = root;
while (cur != nullptr) {
cout << cur->data << " ";

if (!cur->lthread) cur = cur->left;
else if (!cur->rthread) cur = cur->right; else {
while (cur != nullptr && cur->rthread) cur = cur->right;
if (cur != nullptr) cur = cur->right;
}
}
}

int main() {
Node* root = nullptr; root = insert(root, 20); root = insert(root, 10);

root = insert(root, 30); root = insert(root, 5); root = insert(root, 15);

cout << "In-order: "; inorder(root);

cout << "\nPre-order: "; preorder(root);
cout << endl; return 0;
}
Sample Output
In-order: 5 10 15 20 30\nPre-order: 20 10 5 15 30
Time / Space Complexity: Each traversal visits every node once: O(n) time and O(1) extra space.

Assignment 7: Fractional and 0/1 Knapsack
Aim: Solve fractional knapsack using a greedy method and 0/1 knapsack using dynamic programming. The example shows that greedy selection is not always optimal for 0/1 knapsack.
C++17 Code
#include <iostream> #include <algorithm> #include <vector> using namespace std;

struct Item {
int weight, value; double ratio;
};

int main() {
int n = 3, capacity = 50;
Item a[3] = {{10, 60, 6}, {20, 100, 5}, {30, 120, 4}};

// Fractional knapsack: sort by value/weight sort(a, a + n, [](Item x, Item y) {
return x.ratio > y.ratio;
});

int remaining = capacity; double greedyValue = 0;

for (int i = 0; i < n; i++) {
if (a[i].weight <= remaining) { greedyValue += a[i].value; remaining -= a[i].weight;
} else {
greedyValue += a[i].ratio * remaining; break;
}
}

// 0/1 knapsack using dynamic programming int wt[] = {10, 20, 30};
int val[] = {60, 100, 120};
int dp[4][51] = {0};

for (int i = 1; i <= n; i++) {
for (int w = 0; w <= capacity; w++) { dp[i][w] = dp[i - 1][w];

if (wt[i - 1] <= w) { dp[i][w] = max(dp[i][w],
val[i - 1] + dp[i - 1][w - wt[i - 1]]);
}
}
}

cout << "Fractional knapsack value = " << greedyValue << endl; cout << "0/1 knapsack value = " << dp[n][capacity] << endl; return 0;
}
Sample Output
Fractional knapsack value = 240\n0/1 knapsack value = 220\nThe fractional method can take part of an item. The 0
Time / Space Complexity: Fractional greedy: O(n log n). 0/1 dynamic programming: O(nW) time and O(nW) space.

Assignment 8: Bellman-Ford Shortest Path
Aim: Find shortest distances from one source. Bellman-Ford also works with negative edge weights and detects a reachable negative-weight cycle.
C++17 Code
#include <iostream> #include <vector> #include <climits> using namespace std;

struct Edge {
int u, v, w;
};

int main() {
int V = 5, E = 8;
vector<Edge> edges = {
{0, 1, -1}, {0, 2, 4}, {1, 2, 3},
{1, 3, 2}, {1, 4, 2}, {3, 2, 5},
{3, 1, 1}, {4, 3, -3}
};

int source = 0;
vector<int> dist(V, INT_MAX); dist[source] = 0;

// Relax all edges V-1 times for (int i = 1; i < V; i++) {
for (Edge e : edges) {
if (dist[e.u] != INT_MAX && dist[e.u] + e.w < dist[e.v]) {
dist[e.v] = dist[e.u] + e.w;
}
}
}

// Check for negative-weight cycle for (Edge e : edges) {
if (dist[e.u] != INT_MAX && dist[e.u] + e.w < dist[e.v]) { cout << "Negative cycle found"; return 0;
}
}

cout << "Shortest distances from source " << source << ":\n"; for (int i = 0; i < V; i++)
cout << i << " = " << dist[i] << endl;

return 0;
}
Sample Output
Shortest distances from source 0:\n0 = 0\n1 = -1\n2 = 2\n3 = -2\n4 = 1
Time / Space Complexity: O(VE) time because all E edges are checked V-1 times; O(V) extra space for distances.

Assignment 9: N-Queens Using Backtracking
Aim: Place N queens on an N x N chessboard so that no two queens attack each other.
C++17 Code
#include <iostream> #include <cmath> using namespace std;

int n = 4;
int board[20];

bool safe(int row, int col) {
for (int i = 0; i < row; i++) { if (board[i] == col ||
abs(board[i] - col) == abs(i - row)) return false;
}
return true;
}

bool solve(int row) {
if (row == n) return true;

for (int col = 0; col < n; col++) { if (safe(row, col)) {
board[row] = col;

if (solve(row + 1)) return true;
}
}
return false;
}

int main() {
if (solve(0)) {
cout << "Solution:\n";
for (int i = 0; i < n; i++) {
for (int j = 0; j < n; j++) {
if (board[i] == j) cout << "Q "; else cout << ". ";
}
cout << endl;
}
} else {
cout << "No solution";
}
return 0;
}
Sample Output
One valid 4-Queens solution:\n. Q . .\n. . . Q\nQ . . .\n. . Q .
Time / Space Complexity: Worst-case time is approximately O(N!); extra space is O(N) for the queen positions and recursion.

Assignment 10: Travelling Salesperson Problem (Branch and Bound)
Aim: Find a minimum-cost tour that visits every city once and returns to the starting city. A priority queue expands the live node with the smallest lower-bound cost first.
C++17 Code
#include <iostream> #include <queue> #include <vector> #include <climits> using namespace std;

const int N = 4; int cost[N][N] = {
{0, 10, 15, 20},
{10, 0, 35, 25},
{15, 35, 0, 30},
{20, 25, 30, 0}
};

struct Node {
int city, level, costSoFar, bound; vector<int> path;
};

int lowerBound(int city, int level, int costSoFar, bool visited[]) {
int bound = costSoFar;

// Minimum outgoing edge from current city int mn = INT_MAX;
for (int j = 0; j < N; j++)
if (j != city) mn = min(mn, cost[city][j]); bound += mn;

// Minimum outgoing edge for each unvisited city for (int i = 0; i < N; i++) {
if (!visited[i]) { mn = INT_MAX;
for (int j = 0; j < N; j++)
if (j != i) mn = min(mn, cost[i][j]); bound += mn;
}
}
return bound;
}

struct Compare {
bool operator()(Node a, Node b) { return a.bound > b.bound;
}
};

int main() {
priority_queue<Node, vector<Node>, Compare> pq;

bool visited[N] = {true, false, false, false}; Node start;
start.city = 0;
start.level = 1;
start.costSoFar = 0;
start.path = {0};
start.bound = lowerBound(0, 1, 0, visited); pq.push(start);

int bestCost = INT_MAX; vector<int> bestPath;

while (!pq.empty()) { Node cur = pq.top(); pq.pop();
if (cur.bound >= bestCost) continue; if (cur.level == N) {
int total = cur.costSoFar + cost[cur.city][0];
if (total < bestCost) { bestCost = total; bestPath = cur.path; bestPath.push_back(0);
}
continue;
}

for (int next = 1; next < N; next++) {

bool used = false;
for (int x : cur.path)
if (x == next) used = true;

if (!used) {
Node child; child.city = next;
child.level = cur.level + 1; child.costSoFar =
cur.costSoFar + cost[cur.city][next]; child.path = cur.path; child.path.push_back(next);

bool seen[N] = {false};
for (int x : child.path) seen[x] = true;

child.bound = lowerBound(next, child.level,
child.costSoFar, seen); if (child.bound < bestCost) pq.push(child);
}
}
}

cout << "Minimum cost = " << bestCost << endl; cout << "Path: ";
for (int x : bestPath) cout << x << " "; cout << endl;
return 0;
}
Sample Output
Minimum cost = 80\nOne minimum-cost path: 0 1 3 2 0 (another equal-cost tour may also exist).
Time / Space Complexity: Worst case remains O(N!) time. Branch and Bound can reduce the search in practice by pruning paths whose lower bound is not promising.

Assignment 11: Floyd-Warshall All-Pairs Shortest Paths
Aim: Find the shortest distance between every pair of vertices using dynamic programming.
C++17 Code
#include <iostream> using namespace std;
const int INF = 99999; int main() {
int n = 4;
int dist[4][4] = {
{0,	5, INF, 10},
{INF, 0,	3, INF},
{INF, INF, 0,	1},
{INF, INF, INF, 0}
};

// Try every vertex as an intermediate vertex for (int k = 0; k < n; k++) {
for (int i = 0; i < n; i++) {
for (int j = 0; j < n; j++) {
if (dist[i][k] != INF && dist[k][j] != INF && dist[i][k] + dist[k][j] < dist[i][j]) {
dist[i][j] = dist[i][k] + dist[k][j];
}
}
}
}

cout << "Shortest distance matrix:\n"; for (int i = 0; i < n; i++) {
for (int j = 0; j < n; j++) {
if (dist[i][j] == INF) cout << "INF "; else cout << dist[i][j] << "	";
}
cout << endl;
}
return 0;
}
Sample Output
0 5 8 9\nINF 0 3 4\nINF INF 0 1\nINF INF INF 0
Time / Space Complexity: O(V^3) time and O(V^2) space.

Assignment 12: 0/1 Knapsack: Branch and Bound vs Dynamic Programming
Aim: Solve 0/1 knapsack using Branch and Bound and compare its result and general efficiency with dynamic programming.
C++17 Code
#include <iostream> #include <queue> #include <algorithm> #include <vector> using namespace std;

struct Item {
int weight, value; double ratio;
};

Item items[4] = {
{10, 60, 6}, {20, 100, 5},
{30, 120, 4}, {15, 70, 4.67}
};
int n = 4, capacity = 50;

double bound(int level, int weight, int value) { if (weight >= capacity) return value;

double result = value; int totalWeight = weight;

for (int i = level; i < n; i++) {
if (totalWeight + items[i].weight <= capacity) { totalWeight += items[i].weight;
result += items[i].value;
} else {
result += (capacity - totalWeight) * items[i].ratio; break;
}
}
return result;
}

struct Node {
int level, weight, value; double upper;
};

struct Compare {
bool operator()(Node a, Node b) { return a.upper < b.upper;
}
};

int branchBound() {
sort(items, items + n, [](Item a, Item b) { return a.ratio > b.ratio;
});

priority_queue<Node, vector<Node>, Compare> pq; pq.push({0, 0, 0, bound(0, 0, 0)});
int best = 0;

while (!pq.empty()) { Node cur = pq.top(); pq.pop();

if (cur.upper <= best || cur.level == n) continue; Item x = items[cur.level];
// Include current item Node take = {cur.level + 1,
cur.weight + x.weight, cur.value + x.value, 0};

if (take.weight <= capacity) { best = max(best, take.value);
take.upper = bound(take.level, take.weight, take.value); if (take.upper > best) pq.push(take);
}

// Exclude current item
Node skip = {cur.level + 1, cur.weight, cur.value, 0}; skip.upper = bound(skip.level, skip.weight, skip.value);

if (skip.upper > best) pq.push(skip);
}
return best;
}

int dynamicProgramming() { int dp[5][51] = {0};

for (int i = 1; i <= n; i++) {
for (int w = 0; w <= capacity; w++) { dp[i][w] = dp[i - 1][w];
if (items[i - 1].weight <= w) { dp[i][w] = max(dp[i][w],
items[i - 1].value +
dp[i - 1][w - items[i - 1].weight]);
}
}
}
return dp[n][capacity];
}

int main() {
cout << "Branch and Bound result = "
<< branchBound() << endl;
cout << "Dynamic Programming result = "
<< dynamicProgramming() << endl;

cout << "DP time: O(nW); B&B worst case: O(2^n)" << endl; return 0;
}
Sample Output
Branch and Bound result = 230 Dynamic Programming result = 230
DP time: O(nW); B&B worst case: O(2^n)
Time / Space Complexity: Dynamic programming: O(nW) time and O(nW) space. Branch and Bound: O(2^n) worst-case search; pruning may make it faster on some inputs.

























Assignment 03: Adjacency List Using BFS and DFS


#include <iostream>
#include <vector>
#include <queue>
using namespace std;

vector<int> adj[10];
int visited[10];

// BFS Traversal
void BFS(int start)
{
    queue<int> q;
    visited[start] = 1;
    q.push(start);

    cout << "BFS: ";

    while (!q.empty())
    {
        int v = q.front();
        q.pop();

        cout << v << " ";

        for (int i : adj[v])
        {
            if (!visited[i])
            {
                visited[i] = 1;
                q.push(i);
            }
        }
    }
}

// DFS Traversal
void DFS(int v)
{
    visited[v] = 1;
    cout << v << " ";

    for (int i : adj[v])
    {
        if (!visited[i])
        {
            DFS(i);
        }
    }
}

int main()
{
    int n, e, u, v;

    cout << "Enter number of departments: ";
    cin >> n;

    cout << "Enter number of connections: ";
    cin >> e;

    cout << "Enter connections:\n";

    for (int i = 0; i < e; i++)
    {
        cin >> u >> v;

        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    int start;
    cout << "Enter starting department: ";
    cin >> start;

    BFS(start);

    for (int i = 0; i < n; i++)
        visited[i] = 0;

    cout << "\nDFS: ";
    DFS(start);

    return 0;
}
